const colors = [
  { hex: '#e63946', name: 'Crimson' },
  { hex: '#2a9d8f', name: 'Teal' },
  { hex: '#e9c46a', name: 'Saffron' },
  { hex: '#6a4c93', name: 'Violet' },
  { hex: '#f4a261', name: 'Tangerine' },
];
let idx = 0;

function changeColor() {
  idx = (idx + 1) % colors.length;
  const btn = document.getElementById('colorBtn');
  btn.style.background = colors[idx].hex;
  document.getElementById('colorName').textContent
    = colors[idx].name;
}

function showGreeting() {
  const hour = new Date().getHours();
  let msg, icon;

  if (hour >= 5 && hour < 12) {
    msg = 'Good Morning! Rise & shine ☀️';
    icon = '🌅';
  } else if (hour >= 12 && hour < 17) {
    msg = 'Good Afternoon! Hope your day is great 🌤️';
    icon = '🌤️';
  } else if (hour >= 17 && hour < 21) {
    msg = 'Good Evening! Time to unwind 🌇';
    icon = '🌇';
  } else {
    msg = 'Good Night! Sweet dreams 🌙';
    icon = '🌙';
  }

  document.getElementById('greetText').textContent = msg;
  alert(msg); // original alert version
}

function addNumbers() {
  const a = parseFloat(
    document.getElementById('num1').value
  );
  const b = parseFloat(
    document.getElementById('num2').value
  );

  if (isNaN(a) || isNaN(b)) {
    document.getElementById('resultVal')
      .textContent = 'Enter valid numbers!';
    return;
  }

  const sum = a + b;
  document.getElementById('resultVal')
    .textContent = sum;
}
